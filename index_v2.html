        // Calculer la rentabilité

        function calculerRentabilite() {

            currentBien.prix = parseFloat(document.getElementById('prix').value) || 0;

            currentBien.surface = parseFloat(document.getElementById('surface').value) || 0;

            currentBien.loyer = parseFloat(document.getElementById('loyer').value) || 0;

            currentBien.charges = parseFloat(document.getElementById('charges').value) || 0;

            currentBien.ville = document.getElementById('ville').value;

            currentBien.taux = parseFloat(document.getElementById('taux').value) || 0;

            currentBien.duree = parseInt(document.getElementById('duree').value) || 0;

            currentBien.apport = parseFloat(document.getElementById('apport').value) || 0;

            currentBien.categorie = document.getElementById('categorie').value;

            currentBien.dpe = document.getElementById('dpe').value;

            currentBien.travaux = document.getElementById('travaux').value;

            const villeData = villesData[currentBien.ville];

            // Mettre à jour le commentaire DPE

            const dpeComment = dpeComments[currentBien.dpe] || dpeComments['non-renseigne'];

            document.getElementById('dpe-comment').style.display = 'block';

            document.getElementById('dpe-comment-text').textContent = dpeComment;

            const loyerAnnuel = currentBien.loyer * 12;

            const fraisAcquisition = currentBien.prix * 0.08;

            const coutTotal = currentBien.prix + fraisAcquisition;

            // Rentabilité brute

            const rentabiliteBrute = (loyerAnnuel / currentBien.prix * 100).toFixed(2);

            // Mensualité de crédit

            const montantEmprunt = currentBien.prix - currentBien.apport;

            const tauxMensuel = (currentBien.taux / 100) / 12;

            const nombreMois = currentBien.duree * 12;

            const mensualite = montantEmprunt * tauxMensuel * Math.pow(1 + tauxMensuel, nombreMois) /

                             (Math.pow(1 + tauxMensuel, nombreMois) - 1);

            // Cash-flow mensuel

            const cashFlowMensuel = (loyerAnnuel / 12) - mensualite - currentBien.charges;

            // Rentabilité nette (simplifiée)

            const impots = loyerAnnuel * 0.30; // Estimation

            const cashFlowAnnuelNet = loyerAnnuel - (currentBien.charges * 12) - impots;

            const rentabiliteNette = (cashFlowAnnuelNet / coutTotal * 100).toFixed(2);

            // TRI (simplifié)

            const tri = (parseFloat(rentabiliteBrute) * 0.8).toFixed(2);

            // Seuil de rentabilité (breakeven)

            const seuil = (coutTotal / cashFlowAnnuelNet).toFixed(1);

            // Afficher les résultats

            document.getElementById('results').style.display = 'block';

            document.getElementById('rentabilite-brute').textContent = rentabiliteBrute + '%';

            document.getElementById('rentabilite-nette').textContent = rentabiliteNette + '%';

            document.getElementById('cashflow').textContent = cashFlowMensuel.toFixed(0) + ' €';

            document.getElementById('cashflow').className = cashFlowMensuel >= 0 ? 'card-value success' : 'card-value danger';

            document.getElementById('tri').textContent = tri + '%';

            document.getElementById('mensualite').textContent = mensualite.toFixed(0) + ' €';

            document.getElementById('seuil').textContent = seuil + ' ans';

            document.getElementById('prix-m2').textContent = villeData.prixM2 + ' €';

            document.getElementById('tension-locative').textContent = villeData.tensionLocative + '/10';

            document.getElementById('fourchette-loyers').textContent = villeData.fourchetteLoyers.min + ' - ' + villeData.fourchetteLoyers.max + ' €';

            document.getElementById('risque-vacance').textContent = villeData.risqueVacance;

            document.getElementById('ville-name').textContent = villeData.name;

            // Mettre à jour la comparaison

            updateComparison();

            saveBien1();

        }
